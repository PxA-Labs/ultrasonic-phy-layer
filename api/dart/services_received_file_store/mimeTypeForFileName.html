String mimeTypeForFileName(String name) {
  final dot = name.lastIndexOf('.');
  final ext = dot >= 0 ? name.substring(dot + 1).toLowerCase() : '';
  const types = {
    'txt': 'text/plain',
    'csv': 'text/csv',
    'json': 'application/json',
    'pdf': 'application/pdf',
    'png': 'image/png',
    'jpg': 'image/jpeg',
    'jpeg': 'image/jpeg',
    'gif': 'image/gif',
    'wav': 'audio/wav',
    'zip': 'application/zip',
  };
  return types[ext] ?? 'application/octet-stream';
}
