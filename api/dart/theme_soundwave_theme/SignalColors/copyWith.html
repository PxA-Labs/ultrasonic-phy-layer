@override
SignalColors copyWith({
  Color? success,
  Color? warning,
  Color? danger,
  Color? info,
  Color? neutral,
}) {
  return SignalColors(
    success: success ?? this.success,
    warning: warning ?? this.warning,
    danger: danger ?? this.danger,
    info: info ?? this.info,
    neutral: neutral ?? this.neutral,
  );
}
